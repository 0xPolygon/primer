# Diff Between Versions Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a user compare two versions of the same Primer document and see a GitHub-style diff of the rendered text, reachable from the version selector and via a shareable URL.

**Architecture:** Fully client-side. The version selector links to a new picker page that lists every version with `from`/`to` radio columns. Picking two distinct versions navigates to `/d/:docId/compare/:fromV/:toV?view=…`, which fetches both versions' rendered HTML, extracts plaintext, and renders the diff with `react-diff-viewer-continued`. No backend changes, no schema changes.

**Tech Stack:** React 19 + Vite + React Router 7 + TanStack Query 5, Tailwind v4, Vitest (Node env). New runtime dep: `react-diff-viewer-continued`.

**Spec:** `docs/superpowers/specs/2026-06-24-diff-between-versions-design.md` (commit `a550813`)
**Issue:** [#20](https://github.com/0xPolygon/primer/issues/20)

## Global Constraints

- Package manager: **pnpm only**. Never `npm`. Always `--frozen-lockfile` in CI; local installs go through `pnpm install`.
- Supply-chain gate (workspace-level, already in `pnpm-workspace.yaml`): `minimumReleaseAge: 14d`, `allowBuilds: {}` (no install scripts), `blockExoticSubdeps`. Any new dep must clear all three before it lands.
- Module imports inside `apps/web/src` use explicit `.ts` / `.tsx` extensions (e.g. `'../lib/useTheme.ts'`). Match the existing convention in every new file.
- Vitest is configured with `environment: 'node'`. There is no `jsdom`, no `@testing-library/react`, and no React component tests in the repo. Tests in this plan are pure-function unit tests only.
- Types come from `@primer/shared` (`VersionSummaryDTO`, `VersionsListResponse`). Do not duplicate them.
- API calls go through `apps/web/src/lib/fetch.ts` `api()` — never raw `fetch()`. That helper handles CSRF + 401 redirect.
- No backend route additions, no `@primer/shared` changes, no migrations.

Create `apps/web/src/lib/extractText.ts`:

```ts
/**
 * Convert rendered HTML to deterministic plaintext for diffing.
 *
 * Block-level boundaries emit trailing newlines so each paragraph, list
 * item, table row, heading, etc. appears on its own line. <script> and
 * <style> are dropped. <svg> (and descendants) collapses to `[diagram]`
 * — diffing SVG markup is noise.
 */
const BLOCK_TAGS = new Set([
  'P',
  'H1',
  'H2',
  'H3',
  'H4',
  'H5',
  'H6',
  'LI',
  'TR',
  'PRE',
  'BLOCKQUOTE',
  'DIV',
  'SECTION',
  'ARTICLE',
  'FIGURE',
  'HR',
]);

const DROP_TAGS = new Set(['SCRIPT', 'STYLE']);

export function extractText(html: string): string {
  if (html === '') return '';
  const doc = new DOMParser().parseFromString(html, 'text/html');
  const out: string[] = [];
  walk(doc.body, out);
  const joined = out.join('');
  // Collapse 3+ blank lines (\n\n\n...) into exactly 2 blank lines (\n\n\n).
  const collapsed = joined.replace(/\n{4,}/g, '\n\n\n');
  return collapsed.trim();
}

function walk(node: Node, out: string[]): void {
  if (node.nodeType === Node.TEXT_NODE) {
    out.push(node.textContent ?? '');
    return;
  }
  if (node.nodeType !== Node.ELEMENT_NODE) return;
  const el = node as Element;
  const tag = el.tagName.toUpperCase();

  if (DROP_TAGS.has(tag)) return;

  if (tag === 'SVG') {
    out.push('[diagram]\n');
    return;
  }

  if (tag === 'BR') {
    out.push('\n');
    return;
  }

  if (tag === 'PRE') {
    // Preserve internal whitespace verbatim, then a single trailing newline.
    out.push(el.textContent ?? '');
    out.push('\n');
    return;
  }

  for (const child of Array.from(el.childNodes)) {
    walk(child, out);
  }

  if (BLOCK_TAGS.has(tag)) {
    out.push('\n');
  }
}
```

```ts
import type { VersionSummaryDTO } from '@primer/shared';

export type CompareView = 'split' | 'unified';

export function parseViewParam(raw: string | null): CompareView {
  return raw === 'unified' ? 'unified' : 'split';
}

export type CompareResolution =
  | { kind: 'invalid' }
  | { kind: 'not-found'; missing: number }
  | { kind: 'identical'; version: VersionSummaryDTO }
  | { kind: 'ok'; from: VersionSummaryDTO; to: VersionSummaryDTO };

function parsePositiveInt(raw: string | undefined): number | null {
  if (raw === undefined) return null;
  if (!/^[1-9]\d*$/.test(raw)) return null;
  const n = Number(raw);
  return Number.isSafeInteger(n) ? n : null;
}

export function resolveCompareParams(
  versions: VersionSummaryDTO[],
  fromV: string | undefined,
  toV: string | undefined,
): CompareResolution {
  const fromN = parsePositiveInt(fromV);
  const toN = parsePositiveInt(toV);
  if (fromN === null || toN === null) return { kind: 'invalid' };

  const from = versions.find((x) => x.versionNumber === fromN);
  if (!from) return { kind: 'not-found', missing: fromN };

  const to = versions.find((x) => x.versionNumber === toN);
  if (!to) return { kind: 'not-found', missing: toN };

  if (fromN === toN) return { kind: 'identical', version: from };

  return { kind: 'ok', from, to };
}

export function pickDefaultFromVersion(
  versions: VersionSummaryDTO[],
  toVersionNumber: number,
): VersionSummaryDTO | null {
  // Versions arrive in descending order; the "version just below" the target
  // is the next one in the list after the target.
  const idx = versions.findIndex((x) => x.versionNumber === toVersionNumber);
  if (idx === -1) return null;
  return versions[idx + 1] ?? null;
}
```

```tsx
import { useEffect, useMemo, useState } from 'react';
import { Link, useNavigate, useParams } from 'react-router-dom';
import type { VersionSummaryDTO } from '@primer/shared';
import { useVersionsQuery } from '../queries/versions.ts';
import { pickDefaultFromVersion } from '../lib/compareParams.ts';

function relativeTime(iso: string): string {
  const d = new Date(iso);
  const diffMs = Date.now() - d.getTime();
  const sec = Math.floor(diffMs / 1000);
  if (sec < 60) return 'just now';
  const min = Math.floor(sec / 60);
  if (min < 60) return `${min}m ago`;
  const hr = Math.floor(min / 60);
  if (hr < 24) return `${hr}h ago`;
  const day = Math.floor(hr / 24);
  if (day < 7) return `${day}d ago`;
  return d.toLocaleDateString();
}

export default function CompareSelectPage() {
  const { docId = '' } = useParams<{ docId: string }>();
  const navigate = useNavigate();
  const query = useVersionsQuery(docId);

  const versions: VersionSummaryDTO[] = query.data?.versions ?? [];
  const latest = versions[0];
  const defaultFrom = useMemo(
    () => (latest ? pickDefaultFromVersion(versions, latest.versionNumber) : null),
    [versions, latest],
  );

  const [fromV, setFromV] = useState<number | null>(null);
  const [toV, setToV] = useState<number | null>(null);

  // Apply defaults once versions arrive (after first render to avoid setState-in-render).
  // Only fires when `latest` first becomes available; subsequent refetches do not
  // overwrite the user's picks.
  useEffect(() => {
    if (!latest) return;
    if (fromV !== null || toV !== null) return;
    if (defaultFrom) setFromV(defaultFrom.versionNumber);
    setToV(latest.versionNumber);
  }, [latest?.versionNumber, defaultFrom, fromV, toV]);

  const canCompare = fromV !== null && toV !== null && fromV !== toV;

  function onCompare() {
    if (!canCompare) return;
    navigate(`/d/${docId}/compare/${fromV}/${toV}`);
  }

  function setFrom(n: number) {
    setFromV(n);
    if (toV === n) setToV(null);
  }
  function setTo(n: number) {
    setToV(n);
    if (fromV === n) setFromV(null);
  }

  if (query.isLoading) {
    return <div className="p-6 text-sm text-neutral-500">Loading versions…</div>;
  }
  if (query.isError) {
    return (
      <div className="p-6 text-sm text-red-600 dark:text-red-400">
        Failed to load versions.{' '}
        <button
          type="button"
          onClick={() => query.refetch()}
          className="underline"
        >
          Retry
        </button>
      </div>
    );
  }

  if (versions.length < 2) {
    return (
      <div className="mx-auto max-w-2xl p-6">
        <Link to={`/d/${docId}`} className="text-xs text-neutral-500 underline">
          ← Back to document
        </Link>
        <h1 className="mt-4 text-lg font-semibold">Compare versions</h1>
        <p className="mt-4 text-sm text-neutral-600 dark:text-neutral-400">
          This document only has one version. Add another version to compare.
        </p>
      </div>
    );
  }

  return (
    <div className="mx-auto max-w-2xl p-6">
      <Link to={`/d/${docId}`} className="text-xs text-neutral-500 underline">
```

- Produces: a React component `ComparePage` (default export) — registered into the router in Task 6.

- [ ] **Step 1: Implement the result page**

Create `apps/web/src/routes/compare.tsx`:

```tsx
import { useMemo } from 'react';
import { Link, useParams, useSearchParams } from 'react-router-dom';
import { useQuery } from '@tanstack/react-query';
import ReactDiffViewer, { DiffMethod } from 'react-diff-viewer-continued';
import { useVersionsQuery } from '../queries/versions.ts';
import { useTheme } from '../lib/useTheme.ts';
import { api } from '../lib/fetch.ts';
import { extractText } from '../lib/extractText.ts';
import {
  parseViewParam,
  resolveCompareParams,
  type CompareView,
} from '../lib/compareParams.ts';

function useVersionContent(versionId: string | undefined) {
  return useQuery({
    queryKey: ['version-content', versionId ?? ''],
    queryFn: async ({ signal }): Promise<string> => {
      if (!versionId) throw new Error('versionId required');
      const res = await api(`/api/versions/${versionId}/content`, { signal });
      if (!res.ok) throw new Error(`content fetch failed: ${res.status}`);
      return res.text();
    },
    enabled: !!versionId,
    staleTime: 60_000,
  });
}

function diffStylesFromTheme(isDark: boolean) {
  if (isDark) {
    return {
      variables: {
        dark: {
          diffViewerBackground: '#0a0a0a',
          diffViewerColor: '#e5e5e5',
          addedBackground: '#0f3a1f',
          addedColor: '#d1fae5',
          removedBackground: '#3f1a1a',
          removedColor: '#fecaca',
          wordAddedBackground: '#16632e',
          wordRemovedBackground: '#7a1f1f',
          gutterBackground: '#171717',
          gutterColor: '#525252',
          emptyLineBackground: '#0f0f0f',
        },
      },
    };
  }
  return {
    variables: {
      light: {
        diffViewerBackground: '#ffffff',
        diffViewerColor: '#1f2937',
        addedBackground: '#dcfce7',
        addedColor: '#14532d',
        removedBackground: '#fee2e2',
        removedColor: '#7f1d1d',
        wordAddedBackground: '#86efac',
        wordRemovedBackground: '#fca5a5',
        gutterBackground: '#fafafa',
        gutterColor: '#9ca3af',
        emptyLineBackground: '#f9fafb',
      },
    },
  };
}

export default function ComparePage() {
  const { docId = '', fromV, toV } = useParams<{
    docId: string;
    fromV: string;
    toV: string;
  }>();
  const [searchParams, setSearchParams] = useSearchParams();
  const view: CompareView = parseViewParam(searchParams.get('view'));

  const versionsQuery = useVersionsQuery(docId);
  const versions = versionsQuery.data?.versions ?? [];
  const resolution = useMemo(
    () => resolveCompareParams(versions, fromV, toV),
    [versions, fromV, toV],
  );

  const fromId = resolution.kind === 'ok' ? resolution.from.id : undefined;
  const toId = resolution.kind === 'ok' ? resolution.to.id : undefined;
  const fromContent = useVersionContent(fromId);
  const toContent = useVersionContent(toId);

  const { effective } = useTheme();
  const isDark = effective === 'dark';

  function toggleView(next: CompareView) {
    const params = new URLSearchParams(searchParams);
    if (next === 'split') params.delete('view');
    else params.set('view', next);
    setSearchParams(params, { replace: true });
  }

  const header = (
    <div className="flex items-center justify-between gap-4 border-b border-neutral-200 px-6 py-3 dark:border-neutral-800">
      <div className="flex items-center gap-3">
        <Link to={`/d/${docId}`} className="text-xs text-neutral-500 underline">
          ← Back
        </Link>
        {resolution.kind === 'ok' && (
          <span className="font-mono text-sm text-neutral-700 dark:text-neutral-300">
            v{resolution.from.versionNumber} → v{resolution.to.versionNumber}
          </span>
        )}
      </div>
      <div className="inline-flex overflow-hidden rounded border border-neutral-200 text-xs dark:border-neutral-800">
        <button
          type="button"
          onClick={() => toggleView('split')}
          aria-pressed={view === 'split'}
          className={`px-3 py-1 ${
            view === 'split'
              ? 'bg-neutral-900 text-white dark:bg-neutral-100 dark:text-neutral-900'
              : 'bg-white text-neutral-700 dark:bg-neutral-900 dark:text-neutral-300'
          }`}
        >
          Split
        </button>
        <button
          type="button"
          onClick={() => toggleView('unified')}
          aria-pressed={view === 'unified'}
          className={`px-3 py-1 ${
            view === 'unified'
              ? 'bg-neutral-900 text-white dark:bg-neutral-100 dark:text-neutral-900'
              : 'bg-white text-neutral-700 dark:bg-neutral-900 dark:text-neutral-300'
          }`}
        >
          Unified
        </button>
      </div>
    </div>
  );

  if (versionsQuery.isLoading) {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-neutral-500">Loading versions…</div>
      </div>
    );
  }

  if (versionsQuery.isError) {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-red-600 dark:text-red-400">Failed to load versions.</div>
      </div>
    );
  }

  if (resolution.kind === 'invalid') {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-neutral-700 dark:text-neutral-300">
          <p>Invalid compare URL.</p>
          <Link to={`/d/${docId}/compare`} className="mt-2 inline-block underline">
            Pick versions to compare →
          </Link>
        </div>
      </div>
    );
  }

  if (resolution.kind === 'not-found') {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-neutral-700 dark:text-neutral-300">
          <p>Version v{resolution.missing} not found.</p>
          <Link to={`/d/${docId}/compare`} className="mt-2 inline-block underline">
            Pick versions to compare →
          </Link>
        </div>
      </div>
    );
  }

  if (resolution.kind === 'identical') {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-neutral-700 dark:text-neutral-300">
          These two versions render to identical text.
        </div>
      </div>
    );
  }

  if (fromContent.isLoading || toContent.isLoading) {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-neutral-500">Loading diff…</div>
      </div>
    );
  }

  if (fromContent.isError || toContent.isError) {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-red-600 dark:text-red-400">
          <p>Failed to load version content.</p>
          <button
            type="button"
            onClick={() => {
              fromContent.refetch();
              toContent.refetch();
            }}
            className="mt-2 underline"
          >
            Retry
          </button>
        </div>
      </div>
    );
  }

  const fromText = extractText(fromContent.data ?? '');
  const toText = extractText(toContent.data ?? '');

  if (fromText === toText) {
    return (
      <div>
        {header}
        <div className="p-6 text-sm text-neutral-700 dark:text-neutral-300">
          These two versions render to identical text.
        </div>
      </div>
    );
  }

  return (
    <div>
      {header}
      <div className="p-2">
        <ReactDiffViewer
          oldValue={fromText}
          newValue={toText}
          splitView={view === 'split'}
          compareMethod={DiffMethod.WORDS}
          showDiffOnly={false}
          useDarkTheme={isDark}
          leftTitle={`v${resolution.from.versionNumber}`}
          rightTitle={`v${resolution.to.versionNumber}`}
          styles={diffStylesFromTheme(isDark)}
        />
      </div>
    </div>
  );
}
```
